$('bSendGift').addEventListener('click', async () => { if (!activeGiftUser) return; const title = $('giftMailTitle').value.trim(); const body = $('giftMailBody').value.trim(); const badgeId = $('giftBadgeSelect').value; const method = $('giftMethodSelect').value; if (!title || !body) { alert('Please enter both Mail Subject/Title and Message Body!'); return; } $('bSendGift').disabled = true; $('bSendGift').textContent = 'Processing...'; try { const userDocRef = doc(db, 'users', activeGiftUser.id); const mailsColRef = collection(db, 'users', activeGiftUser.id, 'mails'); const mailData = { title: title, body: body, read: false, createdAt: serverTimestamp(), sender: 'Staff Admin' }; if (badgeId !== 'none') { mailData.giftBadgeId = badgeId; if (method === 'direct') { const directPayload = { badges: arrayUnion(badgeId) }; if (badgeId === 'blue_tick') { directPayload.hasBlueTick = true; directPayload.isVerified = true; } await updateDoc(userDocRef, directPayload); mailData.collected = true; mailData.read = false; } else { mailData.collected = false; mailData.read = false; } } await addDoc(mailsColRef, mailData); // In-app notification fallback await addDoc(collection(db, 'notifications'), { userId: activeGiftUser.id, title: "New Mail Received 📩", body: `You have received a new staff message: "${title}"`, type: 'custom', read: false, createdAt: serverTimestamp() }); // 🔥 NEW: Send personal push notification via FCM try { const giftEmoji = badgeId !== 'none' ? '🎁' : '📬'; const notificationSent = await window.sendPersonalNotification(activeGiftUser.id, { title: `New Gift Mail ${giftEmoji}`, body: title, icon: 'https://arenax.cyou/icon-512.png', url: 'https://arenax.cyou/#/mail', data: { type: 'gift_mail', badgeId: badgeId || 'none', from: 'admin_staff' } }); console.log('[Admin Gift] Push notification sent:', notificationSent); } catch (fcmErr) { console.warn('[Admin Gift] Push notification failed (non-blocking):', fcmErr?.message || fcmErr); // Continue - in-app notification already saved above } alert(`Success! Mail sent to ${activeGiftUser.name} successfully. ${badgeId !== 'none' ? `[Badge: ${badgeId} gifted via ${method} grant]` : ''} ✅`); $('mGiftMail').classList.add('hidden'); } catch (err) { alert('Failed to send mail gift: ' + err.message); console.error('[Admin Gift] Error:', err); } finally { $('bSendGift').disabled = false; $('bSendGift').textContent = 'Send Mail & Gift'; } });
